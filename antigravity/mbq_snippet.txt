def setup_bq_dataset():
    client = bigquery.Client(project=PROJECT_ID)
    dataset_ref = bigquery.DatasetReference(PROJECT_ID, DATASET_ID)
    try:
        client.get_dataset(dataset_ref)
        print(f"✅ Dataset '{DATASET_ID}' verified.")
    except NotFound:
        dataset = bigquery.Dataset(dataset_ref)
        dataset.location = LOCATION
        client.create_dataset(dataset)
        print(f"✨ Created new dataset: {PROJECT_ID}.{DATASET_ID}")


def clean_for_bq(value):
    """Recursively cleans a document to ensure strict JSON compatibility."""
    if isinstance(value, dict):
        return {k: clean_for_bq(v) for k, v in value.items()}
    elif isinstance(value, list):
        return [clean_for_bq(v) for v in value]
    elif isinstance(value, datetime):
        return value.isoformat()
    elif type(value).__name__ == 'ObjectId':  # Safely handle Mongo ObjectIds
        return str(value)
    else:
        return value

                "isBot": False,
                "browserType": {"$ne": "Unknown"},
                "deviceType": {"$ne": "Unknown"},
                "isFromGoogle": True
            }

        row_count = 0
